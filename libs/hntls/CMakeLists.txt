cmake_minimum_required(VERSION 3.23)
include(CMakePackageConfigHelpers)
add_library(hntls src/session.cpp src/stream.cpp src/client.cpp)
add_library(hntls::hntls ALIAS hntls)
target_compile_features(hntls PUBLIC cxx_std_20)
set_target_properties(hntls PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(hntls PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hntls PUBLIC asyncpp socketpp::socketpp certpp::certpp)
option(HNTLS_BUILD_EXAMPLES "Build standalone TLS loopback interoperability examples" ON)
if(HNTLS_BUILD_EXAMPLES)
    add_executable(hntls_echo_server examples/echo_server.cpp)
    add_executable(hntls_echo_client examples/echo_client.cpp)
    target_link_libraries(hntls_echo_server PRIVATE hntls::hntls)
    target_link_libraries(hntls_echo_client PRIVATE hntls::hntls)
endif()
if(BUILD_SHARED_LIBS)
    target_compile_definitions(hntls PUBLIC HNTLS_SHARED PRIVATE HNTLS_BUILDING_LIBRARY)
endif()
if(HNUTILS_BUILD_TESTS)
    add_executable(hntls_tests tests/main.cpp tests/contracts.cpp tests/transport.cpp)
    target_link_libraries(hntls_tests PRIVATE hntls::hntls socketpp::socketpp doctest::doctest)
    # Each public header must compile first, without the umbrella or incidental include order.
    foreach(HNTLS_HEADER IN ITEMS hntls.hpp hntls/common.hpp hntls/options.hpp
            hntls/credentials.hpp hntls/context.hpp hntls/session.hpp hntls/stream.hpp
            hntls/client.hpp hntls/socket_stream.hpp)
        string(MAKE_C_IDENTIFIER "${HNTLS_HEADER}" HNTLS_HEADER_ID)
        set(HNTLS_HEADER_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/header-check/${HNTLS_HEADER_ID}.cpp")
        file(GENERATE OUTPUT "${HNTLS_HEADER_SOURCE}" CONTENT "#include <${HNTLS_HEADER}>\n")
        target_sources(hntls_tests PRIVATE "${HNTLS_HEADER_SOURCE}")
    endforeach()
    include("${DOCTEST_CMAKE_HELPER}")
    doctest_discover_tests(hntls_tests TEST_PREFIX "hntls." PROPERTIES TIMEOUT 30)
endif()
install(TARGETS hntls EXPORT hntls-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT hntls-targets NAMESPACE hntls:: DESTINATION lib/cmake/hntls)
configure_package_config_file(cmake/hntls-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hntls-config.cmake" INSTALL_DESTINATION lib/cmake/hntls)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hntls-config-version.cmake" VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hntls-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hntls-config-version.cmake" DESTINATION lib/cmake/hntls)
