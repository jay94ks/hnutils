cmake_minimum_required(VERSION 3.23)
include(CMakePackageConfigHelpers)

add_library(hnhttp src/body.cpp src/http1.cpp src/endpoint.cpp src/header_codec.cpp src/huffman.cpp src/qpack.cpp)
target_sources(hnhttp PRIVATE src/multipart_source.cpp src/form_reader.cpp)
target_sources(hnhttp PRIVATE src/multiplexed.cpp src/http2.cpp src/http3.cpp src/native_transport.cpp)
target_sources(hnhttp PRIVATE src/secure_endpoint.cpp src/resource.cpp)
add_library(hnhttp::hnhttp ALIAS hnhttp)
target_compile_features(hnhttp PUBLIC cxx_std_20)
set_target_properties(hnhttp PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(hnhttp PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hnhttp PUBLIC asyncpp socketpp::socketpp hntls::hntls hnquic::hnquic)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(hnhttp PUBLIC HNHTTP_SHARED PRIVATE HNHTTP_BUILDING_LIBRARY)
endif()
if(MSVC)
    target_compile_options(hnhttp PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(hnhttp PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        # GCC diagnoses asyncpp's paired coroutine allocator overloads as mismatched.
        target_compile_options(hnhttp PRIVATE -Wno-mismatched-new-delete)
    endif()
endif()

if(HNUTILS_BUILD_TESTS)
    set(header_sources)
    foreach(header common fields transport body form message connection context endpoint header_codec qpack resource)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.cpp")
        file(GENERATE OUTPUT "${source}" CONTENT "#include <hnhttp/${header}.hpp>\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(hnhttp_header_checks OBJECT ${header_sources} examples/http1.cpp examples/multiplexed.cpp
        examples/socketpp_reactor.cpp examples/plaintext.cpp examples/native_transport.cpp examples/secure_client.cpp)
    target_link_libraries(hnhttp_header_checks PRIVATE hnhttp::hnhttp socketpp::socketpp)
    set_target_properties(hnhttp_header_checks PROPERTIES CXX_EXTENSIONS OFF)
    add_executable(hnhttp_contract_tests tests/main.cpp tests/contracts.cpp tests/reactor.cpp tests/http1.cpp tests/codecs.cpp tests/multiplexed.cpp tests/forms.cpp tests/protocol_errors.cpp tests/native_engines.cpp tests/secure_endpoints.cpp tests/resource.cpp)
    target_link_libraries(hnhttp_contract_tests PRIVATE hnhttp::hnhttp doctest::doctest socketpp::socketpp)
    add_dependencies(hnhttp_contract_tests hnhttp_header_checks)
    add_executable(hnhttp_http2_server examples/http2_server.cpp)
    target_link_libraries(hnhttp_http2_server PRIVATE hnhttp::hnhttp)
    add_executable(hnhttp_secure_server examples/secure_server.cpp)
    target_link_libraries(hnhttp_secure_server PRIVATE hnhttp::hnhttp)
    include("${DOCTEST_CMAKE_HELPER}")
    doctest_discover_tests(hnhttp_contract_tests TEST_PREFIX "hnhttp.contract." PROPERTIES TIMEOUT 30)
endif()

install(TARGETS hnhttp EXPORT hnhttp-targets)
install(DIRECTORY include/ DESTINATION include)
install(FILES RFC-CODE-LICENSE.txt DESTINATION share/licenses/hnhttp)
install(EXPORT hnhttp-targets NAMESPACE hnhttp:: DESTINATION lib/cmake/hnhttp)
configure_package_config_file(cmake/hnhttp-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hnhttp-config.cmake"
    INSTALL_DESTINATION lib/cmake/hnhttp)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hnhttp-config-version.cmake"
    VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hnhttp-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hnhttp-config-version.cmake"
    DESTINATION lib/cmake/hnhttp)
