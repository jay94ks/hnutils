cmake_minimum_required(VERSION 3.23)
include(CMakePackageConfigHelpers)
add_library(hnquic src/endpoint.cpp)
add_library(hnquic::hnquic ALIAS hnquic)
target_compile_features(hnquic PUBLIC cxx_std_20)
set_target_properties(hnquic PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(hnquic PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hnquic PUBLIC asyncpp hntls::hntls socketpp::socketpp)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(hnquic PUBLIC HNQUIC_SHARED PRIVATE HNQUIC_BUILDING_LIBRARY)
endif()
if(HNUTILS_BUILD_TESTS)
    add_executable(hnquic_tests tests/main.cpp tests/contracts.cpp)
    target_link_libraries(hnquic_tests PRIVATE hnquic::hnquic socketpp::socketpp doctest::doctest)
    # Compile individual public contracts independently of the umbrella header.
    foreach(HNQUIC_HEADER IN ITEMS hnquic.hpp hnquic/common.hpp hnquic/options.hpp
            hnquic/stream.hpp hnquic/connection.hpp hnquic/server.hpp hnquic/client.hpp)
        string(MAKE_C_IDENTIFIER "${HNQUIC_HEADER}" HNQUIC_HEADER_ID)
        set(HNQUIC_HEADER_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/header-check/${HNQUIC_HEADER_ID}.cpp")
        file(GENERATE OUTPUT "${HNQUIC_HEADER_SOURCE}" CONTENT "#include <${HNQUIC_HEADER}>\n")
        target_sources(hnquic_tests PRIVATE "${HNQUIC_HEADER_SOURCE}")
    endforeach()
    include("${DOCTEST_CMAKE_HELPER}")
    doctest_discover_tests(hnquic_tests TEST_PREFIX "hnquic." PROPERTIES TIMEOUT 30)
endif()
install(TARGETS hnquic EXPORT hnquic-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT hnquic-targets NAMESPACE hnquic:: DESTINATION lib/cmake/hnquic)
configure_package_config_file(cmake/hnquic-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hnquic-config.cmake" INSTALL_DESTINATION lib/cmake/hnquic)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hnquic-config-version.cmake" VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hnquic-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hnquic-config-version.cmake" DESTINATION lib/cmake/hnquic)
