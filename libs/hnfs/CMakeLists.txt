cmake_minimum_required(VERSION 3.23)
include(CMakePackageConfigHelpers)

add_library(hnfs src/common.cpp src/device.cpp src/file_device.cpp src/filesystem.cpp
    src/extensions.cpp src/cpio.cpp src/indexed_archive.cpp src/async_archive.cpp src/async_tree.cpp
    src/partition.cpp src/optical.cpp src/virtual_disk.cpp)
add_library(hnfs::hnfs ALIAS hnfs)
target_compile_features(hnfs PUBLIC cxx_std_20)
set_target_properties(hnfs PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(hnfs PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hnfs PUBLIC asyncpp)
if(UNIX)
    target_compile_definitions(hnfs PRIVATE _FILE_OFFSET_BITS=64)
endif()
if(BUILD_SHARED_LIBS)
    target_compile_definitions(hnfs PUBLIC HNFS_SHARED PRIVATE HNFS_BUILDING_LIBRARY)
endif()
if(MSVC)
    target_compile_options(hnfs PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(hnfs PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(hnfs PRIVATE -Wno-mismatched-new-delete)
    endif()
endif()

if(HNUTILS_BUILD_TESTS)
    add_executable(hnfs_tests tests/main.cpp tests/filesystem_tests.cpp tests/disk_tests.cpp)
    target_link_libraries(hnfs_tests PRIVATE hnfs::hnfs doctest::doctest)
    include("${DOCTEST_CMAKE_HELPER}")
    doctest_discover_tests(hnfs_tests TEST_PREFIX "hnfs." PROPERTIES TIMEOUT 60)
    add_executable(hnfs_oracle tests/oracle.cpp)
    target_link_libraries(hnfs_oracle PRIVATE hnfs::hnfs)
    add_executable(hnfs_disk_oracle tests/disk_oracle.cpp)
    target_link_libraries(hnfs_disk_oracle PRIVATE hnfs::hnfs)
    option(HNFS_DISK_REFERENCE_TESTS "Run test-only genisoimage, sfdisk, sgdisk and qemu-img interoperability" OFF)
    if(HNFS_DISK_REFERENCE_TESTS)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        foreach(program genisoimage sfdisk sgdisk qemu-img)
            find_program(HNFS_TEST_${program} ${program} REQUIRED)
        endforeach()
        add_test(NAME hnfs.disk.reference COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/disk_reference_check.py" "$<TARGET_FILE:hnfs_disk_oracle>")
        set_tests_properties(hnfs.disk.reference PROPERTIES TIMEOUT 120 LABELS reference ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    endif()
    option(HNFS_REFERENCE_TESTS "Run optional test-only GNU cpio interoperability" OFF)
    if(HNFS_REFERENCE_TESTS)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        find_program(HNFS_CPIO_TEST cpio REQUIRED)
        add_test(NAME hnfs.reference COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_check.py" "$<TARGET_FILE:hnfs_oracle>" "${HNFS_CPIO_TEST}")
        set_tests_properties(hnfs.reference PROPERTIES TIMEOUT 60 LABELS reference ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    endif()
endif()

option(HNFS_BUILD_EXAMPLES "Build hnfs examples" OFF)
if(HNFS_BUILD_EXAMPLES)
    add_executable(hnfs_cpio_example examples/cpio.cpp)
    target_link_libraries(hnfs_cpio_example PRIVATE hnfs::hnfs)
    if(HNUTILS_BUILD_TESTS)
        add_test(NAME hnfs.example COMMAND hnfs_cpio_example)
        set_tests_properties(hnfs.example PROPERTIES TIMEOUT 30)
    endif()
endif()

install(TARGETS hnfs EXPORT hnfs-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin INCLUDES DESTINATION include)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT hnfs-targets NAMESPACE hnfs:: DESTINATION lib/cmake/hnfs)
configure_package_config_file(cmake/hnfs-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hnfs-config.cmake"
    INSTALL_DESTINATION lib/cmake/hnfs)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hnfs-config-version.cmake"
    VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hnfs-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hnfs-config-version.cmake"
    DESTINATION lib/cmake/hnfs)
